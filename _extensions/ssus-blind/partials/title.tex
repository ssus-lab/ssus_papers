% title.tex for ssus-blind — quarto's stock title block, adapted.
%
% Two differences from the stock block, and the first is the format's whole
% stance: the BLIND COPY OMITS the identity pieces by construction.
%   1. No author emission — not even conditionally. A project whose config
%      carries the real author block is SAFE here: this format never asks
%      for it, so the block's presence cannot leak. (The block still feeds
%      the display copy and the title page, which read the same config.)
%   2. The thanks splice is GONE: quarto's stock title.tex folds the
%      acknowledgments into \title, which de-blinds the manuscript; along
%      with the header's no-op \thanks backstop, an acknowledgments string
%      in the config never reaches the blind output either.
$if(title)$
\title{$title$}
$endif$
$if(subtitle)$
\makeatletter
\providecommand{\subtitle}[1]{%
  \apptocmd{\@title}{\par #1 \par}{}{}
}
\makeatother
\subtitle{$subtitle$}
$endif$
\date{$date$}
