% -----------------
% References
% -----------------
% The hikmah core carries no bibliography styling, so citeproc entries
% land in pandoc's CSLReferences list, whose open spec sets itemsep to
% entry-spacing × \baselineskip — one full blank line between entries.
% An etoolbox AtBeginEnvironment hook cannot fix it (it fires before
% the open code, which overwrites it), so redefine the environment:
% this header loads after pandoc's definitions, hanging indent kept,
% entries set tight as typeset lines.
\ifcsname CSLReferences\endcsname
  \renewenvironment{CSLReferences}[2] % #1 hanging-indent, #2 entry-spacing
   {\begin{list}{}{%
      \setlength{\itemindent}{0pt}
      \setlength{\leftmargin}{0pt}
      \setlength{\parsep}{0pt}
      \ifodd #1
       \setlength{\leftmargin}{\cslhangindent}
       \setlength{\itemindent}{-1\cslhangindent}
      \fi
      \setlength{\itemsep}{2pt}}}
   {\end{list}}
\fi
